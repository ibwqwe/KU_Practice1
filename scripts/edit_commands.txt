"""
Скрипт для тестирования команд touch и rm.
Проверяет создание, удаление и обработку ошибок.
Все изменения — только в памяти. JSON-файл на диске не меняется.

--- начальное содержимое корня ---
"""
ls

"""--- создание файла в корне ---"""
touch newfile.txt
ls

""" --- файл пустой ---"""
cat newfile.txt

"""--- touch: файл уже существует ---"""
touch newfile.txt

"""--- создание файла во вложенной папке ---"""
touch home/newdoc.md
ls home
cat home/newdoc.md

"""--- удаление файла из корня ---"""
rm newfile.txt
ls

"""--- удаление файла из вложенной папки ---"""
rm home/newdoc.md
ls home

"""--- ошибки ---"""

"""# удаление несуществующего файла"""
rm missing.txt

"""# удаление папки"""
rm home

"""# создание файла в несуществующей папке"""
touch missing_folder/file.txt

"""# touch без аргументов"""
touch

"""# rm без аргументов"""
rm

"""# --- переход и создание/удаление во вложенной директории ---"""
cd home
touch temp.txt
ls
rm temp.txt
ls

"""# --- вернуться в корень ---"""
cd /
ls
